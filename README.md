# 1. Create a changeset after making changes

pnpm changeset

# 2. Check what will be versioned

pnpm changeset status

# 3. Consume changesets and bump versions

pnpm changeset version

# 4. Publish to registry (or dry-run)

pnpm changeset publish --dry-run

# 5. Create changeset manually (bypasses auto-detection)

pnpm changeset --empty
